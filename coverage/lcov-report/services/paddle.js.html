/**
 * Paddle Billing client and helpers
 *
 * Provides:
 *   - paddle       — Paddle SDK instance
 *   - PRICE_IDS    — maps SubscriptionTier → Paddle Price ID (set via env vars)
 *   - trackUsage() — upsert a UsageMetric row for the current calendar month
 *   - syncSubscriptionFromPaddle() — write a Paddle subscription event to DB
 *   - syncTransactionFromPaddle()  — write a Paddle transaction event to DB
 */
 
import { Paddle, Environment } from '@paddle/paddle-node-sdk';
import { prisma } from '../db/prisma.ts';
import { createLogger } from '../api/utils/logger.js';
 
const logger = createLogger('PADDLE');
 
if (!process.env.PADDLE_API_KEY) {
  logger.warn('PADDLE_API_KEY not set — billing features will be unavailable');
}
 
export const paddle = process.env.PADDLE_API_KEY
  ? new Paddle(process.env.PADDLE_API_KEY, {
      environment: process.env.PADDLE_ENVIRONMENT === 'production'
        ? Environment.production
        : Environment.sandbox,
    })
  : null;
 
export const PRICE_IDS = {
  BASIC:      process.env.PADDLE_PRICE_BASIC,
  PRO:        process.env.PADDLE_PRICE_PRO,
  ENTERPRISE: process.env.PADDLE_PRICE_ENTERPRISE,
};
 
export function tierFromPriceId(priceId) {
  return Object.entries(PRICE_IDS).find(([, pid]) => pid === priceId)?.[0] ?? 'BASIC';
}
 
// ─────────────────────────────────────────────────────────────────────────────
// Usage tracking — call from route handlers after successful operations.
// Safe to fire-and-forget — errors are logged but never rethrown.
// ─────────────────────────────────────────────────────────────────────────────
 
export async function trackUsage(orgId, field, by = 1) {
  if (!orgId || !field) return;
  const month = new Date();
  month.setUTCDate(1);
  month.setUTCHours(0, 0, 0, 0);
 
  await prisma.usageMetric.upsert({
    where:  { orgId_month: { orgId, month } },
    create: { orgId, month, [field]: by },
    update: { [field]: { increment: by } },
  }).catch((err) => {
    logger.error(`Failed to track usage (${field}) for org ${orgId}: ${err.message}`);
  });
}

// DB sync helpers — called from the Paddle webhook handler
// ─────────────────────────────────────────────────────────────────────────────
 
const SUB_STATUS_MAP = {
  active:   'ACTIVE',
  trialing: 'ACTIVE',
  paused:   'ACTIVE',
  past_due: 'PAST_DUE',
  canceled: 'CANCELLED',
};
 
/**
 * Upsert a Subscription row from a Paddle subscription object.
 * Looks up the org by Paddle customer ID stored in our DB.
 */
export async function syncSubscriptionFromPaddle(paddleSub) {
  const customerId = paddleSub.customerId;
  if (!customerId) return;
 
  const existing = await prisma.subscription.findFirst({ where: { customerId } });
  if (!existing) {
    logger.warn(`syncSubscription: no org found for Paddle customer ${customerId}`);
    return;
  }
 
  const priceId = paddleSub.items?.[0]?.price?.id;
  const tier    = priceId ? tierFromPriceId(priceId) : existing.tier;
 
  const periodStart = paddleSub.currentBillingPeriod?.startsAt
    ? new Date(paddleSub.currentBillingPeriod.startsAt) : new Date();
  const periodEnd   = paddleSub.currentBillingPeriod?.endsAt
    ? new Date(paddleSub.currentBillingPeriod.endsAt)   : new Date(Date.now() + 30 * 86400000);
 
  await prisma.subscription.update({
    where: { id: existing.id },
    data: {
      subscriptionId:     paddleSub.id,
      tier,
      status:             SUB_STATUS_MAP[paddleSub.status] ?? 'ACTIVE',
      currentPeriodStart: periodStart,
      currentPeriodEnd:   periodEnd,
      renewalDate:        periodEnd,
      cancelledAt:        paddleSub.scheduledChange?.action === 'cancel'
                            ? new Date(paddleSub.scheduledChange.effectiveAt) : null,
    },
  }).catch((err) => logger.error(`Failed to sync subscription ${paddleSub.id}: ${err.message}`));
}
 
/**
 * Upsert an Invoice row from a Paddle transaction object.
 * Only processes completed/billed transactions.
 */
export async function syncTransactionFromPaddle(txn) {
  if (!txn.subscriptionId) return;
 
  const sub = await prisma.subscription.findFirst({
    where: { subscriptionId: txn.subscriptionId },
  });
  if (!sub) {
    logger.warn(`syncTransaction: no subscription found for Paddle sub ${txn.subscriptionId}`);
    return;
  }
 
  const txnStatusMap = {
    completed: 'PAID',
    billed:    'SENT',
    canceled:  'REFUNDED',
    past_due:  'FAILED',
  };
 
  const total    = txn.details?.totals?.total;
  const amount   = total ? Math.round(parseFloat(total) * 100) : 0; // Paddle returns decimal strings
  const currency = (txn.currencyCode ?? 'USD').toUpperCase();
  const status   = txnStatusMap[txn.status] ?? 'DRAFT';
 
  await prisma.invoice.upsert({
    where:  { externalInvoiceId: txn.id },
    create: {
      subscriptionId:    sub.id,
      externalInvoiceId: txn.id,
      amount,
      currency,
      status,
      paidAt:  status === 'PAID' ? new Date() : null,
      dueDate: new Date(),
    },
    update: {
      status,
      amount,
      paidAt: status === 'PAID' ? new Date() : null,
    },
  }).catch((err) => logger.error(`Failed to sync transaction ${txn.id}: ${err.message}`));
}
